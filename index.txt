<script>
  let currentMed = '感冒藥';
  let currentInterval = 4; // 預設4小時

  // 選擇藥品
  function selectMed(medName) {
    currentMed = medName;
    document.querySelectorAll('.med-btn').forEach(btn => {
      btn.classList.remove('active');
      if(btn.innerText === medName) btn.classList.add('active');
    });
    updateStatusDisplay();
  }

  // 選擇頻率
  function selectInterval(hours) {
    currentInterval = hours;
    document.querySelectorAll('.time-btn').forEach(btn => {
      btn.classList.remove('active');
      if(btn.innerText.includes(hours)) btn.classList.add('active');
    });
    updateStatusDisplay();
  }

  // 按下打卡
  function punchMedicine() {
    const now = new Date();
    const record = {
      time: now.getTime(),
      interval: currentInterval
    };
    // 儲存到瀏覽器
    localStorage.setItem('med_' + currentMed, JSON.stringify(record));
    updateStatusDisplay();
  }

  // 更新畫面狀態
  function updateStatusDisplay() {
    const recordStr = localStorage.getItem('med_' + currentMed);
    const statusDiv = document.getElementById('status');

    if (!recordStr) {
      statusDiv.innerHTML = `【${currentMed}】<br>目前沒有打卡紀錄喔！`;
      return;
    }

    const record = JSON.parse(recordStr);
    const lastTakenTime = new Date(record.time);
    
    // 計算下次吃藥時間
    const nextTime = new Date(record.time + record.interval * 60 * 60 * 1000);
    const now = new Date();

    let timeText = `上次吃藥：${lastTakenTime.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})}<br>`;
    timeText += `下次預計：${nextTime.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})}<br><br>`;

    // 檢查是否超時或太早吃
    if (now >= nextTime) {
      timeText += `<span class="alert">⚠️ 已經超過預計時間了，可以吃藥囉！</span>`;
    } else {
      timeText += `<span class="ok">✅ 時間還沒到，請稍候再吃。</span>`;
    }

    statusDiv.innerHTML = `【${currentMed}】<br>` + timeText;
  }

  // 網頁載入時先更新一次畫面
  window.onload = updateStatusDisplay;
  
  // 每分鐘自動更新一次狀態檢查（避免網頁一直開著狀態沒變）
  setInterval(updateStatusDisplay, 60000);
</script>
